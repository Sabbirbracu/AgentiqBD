 * Plain SVG + CSS, so it's a server component with no JS cost. Tall elements (chimneys, towers)
 * are kept outside the text column (roughly x 420–1180) so they never sit behind copy.

function rng(seed: number) {
  let s = seed;
  return () => (s = (s * 16807) % 2147483647) / 2147483647;
}

    horizonCity.push([x, w, h]);
    x += w + 2 + Math.floor(r() * 6);
  }
}
const rightCity: [number, number, number][] = [];

function tallest(list: [number, number, number][], from: number, to: number, base: number) {
  const inRange = list.filter(([x, w]) => x >= from && x + w <= to);
  const [x, w, h] = inRange.reduce((a, b) => (b[2] > a[2] ? b : a));
  return { x: x + w / 2, y: base - h - (h > 150 ? 22 : 0) };
}

// a spoke hangs from the core like a cable: sagging below the straight line, so it never rises into the copy
function spoke(n: Node) {
  const dx = n.x - HUB.x;
  const dy = n.y - HUB.y;
  const dist = Math.hypot(dx, dy);
  const mx = HUB.x + dx / 2;
  const my = HUB.y + dy / 2;
  const sag = n.y > HUB.y ? -dist * 0.1 : dist * 0.2;
  return `M${HUB.x} ${HUB.y} Q${mx} ${my + sag} ${n.x} ${n.y}`;
}

function Palm({ x, y, s = 1, lean = 14 }: { x: number; y: number; s?: number; lean?: number }) {
  const leaf = 'M0 0 C14 -17 42 -16 60 5 C42 -6 16 -4 0 0 Z';
  const spread = [-62, -32, -4, 22];
  return (
    <g transform={`translate(${x} ${y}) scale(${s})`} fill={DARK}>
      <path d={`M-2 0 Q${lean * 0.15} -46 ${lean} -88`} stroke={DARK} strokeWidth="5" fill="none" strokeLinecap="round" />
      <g transform={`translate(${lean} -88)`}>
        {spread.map((a) => (
          <g key={a}>
            <path d={leaf} transform={`rotate(${a})`} />
            <path d={leaf} transform={`scale(-1 1) rotate(${a})`} />
          </g>
        ))}
      </g>
    </g>
  );
}

function ripple(y: number, x0: number, n: number) {
  let d = `M${x0} ${y} q22 -5 44 0`;
  for (let i = 0; i < n; i++) d += ' t44 0';
  return d;
}

// banks are listed far → near; a Catmull-Rom pass turns them into smooth curves so the river meanders
const leftBank: [number, number][] = [[745, 566], [752, 606], [742, 648], [640, 698], [450, 758], [260, 830], [140, 900]];
const rightBank: [number, number][] = [[835, 566], [908, 606], [982, 648], [968, 700], [964, 760], [1034, 832], [1190, 900]];

function curve(points: [number, number][]) {
  let d = '';
  for (let i = 0; i < points.length - 1; i++) {
    const p0 = points[i - 1] ?? points[i];
    const p1 = points[i];
    const p2 = points[i + 1];
    const p3 = points[i + 2] ?? p2;
    const c1 = [p1[0] + (p2[0] - p0[0]) / 6, p1[1] + (p2[1] - p0[1]) / 6];
    const c2 = [p2[0] - (p3[0] - p1[0]) / 6, p2[1] - (p3[1] - p1[1]) / 6];
    d += ` C${c1[0]} ${c1[1]} ${c2[0]} ${c2[1]} ${p2[0]} ${p2[1]}`;
  }
  return d;
}
const RIVER =
  `M${leftBank[0][0]} ${leftBank[0][1]}` +
  curve(leftBank) +
  ` L${rightBank[rightBank.length - 1][0]} ${rightBank[rightBank.length - 1][1]}` +
  curve([...rightBank].reverse()) +
  ' Z';
